'use client';

import { useEffect, useRef, useState } from 'react';
import Script from 'next/script';
// On garde ces imports pour le typage et le loader, mais on ne les utilise pas pour le rendu global
import * as THREE from 'three'; 
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader.js';

export default function ARPage() {
  const containerRef = useRef<HTMLDivElement>(null);
  const [status, setStatus] = useState('Chargement des modules AR...');
  const [ready, setReady] = useState(false);

  useEffect(() => {
    if (!ready) return;

    let mindarThree: any = null;

    const startAR = async () => {
      try {
        // On attend que l'objet global soit rattaché par le module
        if (!window.MINDAR?.IMAGE?.MindARThree) {
          setTimeout(startAR, 500);
          return;
        }

        setStatus('Démarrage de la caméra...');

        mindarThree = new window.MINDAR.IMAGE.MindARThree({
          container: containerRef.current!,
          imageTargetSrc: '/models/card.mind',
          maxTrack: 1,
          uiLoading: 'yes', uiScanning: 'yes', uiError: 'yes',
        });

        const { renderer, scene, camera } = mindarThree;
        
        // Correction couleur (valeur numérique pour compatibilité r128/r160)
        renderer.outputEncoding = 3001; 

        const light = new THREE.HemisphereLight(0xffffff, 0xbbbbff, 1);
        scene.add(light);

        const anchor = mindarThree.addAnchor(0);
        const loader = new GLTFLoader();
        
        loader.load('/models/scene.gltf', (gltf) => {
          anchor.group.add(gltf.scene);
          setStatus('VintError : Prêt ! Scannez une bouteille');
        });

        await mindarThree.start();
        renderer.setAnimationLoop(() => renderer.render(scene, camera));

      } catch (err) {
        console.error("Erreur d'init:", err);
        setStatus("Erreur caméra ou navigateur");
      }
    };

    startAR();

    return () => {
      if (mindarThree) mindarThree.stop();
    };
  }, [ready]);

  return (
    <div style={{ width: '100%', height: '100vh', position: 'relative', background: 'black' }}>
      
      {/* 1. CONFIGURATION DE L'IMPORT MAP (Version complète) */}
      <script type="importmap" dangerouslySetInnerHTML={{
        __html: JSON.stringify({
          imports: {
            "three": "https://unpkg.com/three@0.160.0/build/three.module.js",
            "three/addons/renderers/CSS3DRenderer.js": "https://unpkg.com/three@0.160.0/examples/jsm/renderers/CSS3DRenderer.js"
          }
        })
      }} />

      {/* 2. CHARGEMENT DES SCRIPTS EN TANT QUE MODULES */}
      {/* On ajoute crossorigin pour régler l'erreur "request credentials mode" */}
      <Script 
        src="https://unpkg.com/three@0.160.0/build/three.module.js"
        strategy="afterInteractive"
        type="module"
        crossOrigin="anonymous"
      />

      <Script 
        src="https://cdn.jsdelivr.net/npm/mind-ar@1.2.5/dist/mindar-image-three.prod.js"
        strategy="afterInteractive"
        type="module"
        crossOrigin="anonymous"
        onLoad={() => {
          console.log("✅ MindAR chargé");
          setReady(true);
        }}
      />

      <div ref={containerRef} style={{ width: '100%', height: '100%' }} />

      <div style={{
        position: 'absolute', top: '20px', left: '50%', transform: 'translateX(-50%)',
        padding: '12px 24px', background: 'rgba(0,0,0,0.8)', color: 'white', 
        borderRadius: '8px', zIndex: 1000, textAlign: 'center'
      }}>
        {status}
      </div>
    </div>
  );
}
